# Pocket is available only for the ARM64 devices on which it has been validated.
if(NOT ANDROID_ABI STREQUAL "arm64-v8a")
    return()
endif()
set(SPM_ENABLE_SHARED OFF CACHE BOOL "" FORCE)
set(SPM_ENABLE_TCMALLOC OFF CACHE BOOL "" FORCE)
set(SPM_ENABLE_NFKC_COMPILE OFF CACHE BOOL "" FORCE)
set(SPM_BUILD_TEST OFF CACHE BOOL "" FORCE)
set(SPM_PROTOBUF_PROVIDER "internal" CACHE STRING "" FORCE)
set(SPM_ABSL_PROVIDER "internal" CACHE STRING "" FORCE)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
add_subdirectory(sentencepiece EXCLUDE_FROM_ALL)
target_compile_options(sentencepiece-static PRIVATE -w -fvisibility=hidden)
# AGP discovers the upstream CLI targets even with EXCLUDE_FROM_ALL. Their
# protobuf logger needs Android's log library too.
foreach(spm_tool spm_decode spm_encode spm_export_vocab spm_normalize spm_train)
    if(TARGET ${spm_tool})
        target_link_libraries(${spm_tool} log)
    endif()
endforeach()
set(ORT_LIBRARY "${CMAKE_CURRENT_SOURCE_DIR}/../../../../build/pocket-ort/jni/${ANDROID_ABI}/libonnxruntime.so")
if(NOT EXISTS "${ORT_LIBRARY}")
    message(FATAL_ERROR "Run Gradle preparePocketOrt before configuring Pocket native code")
endif()
add_library(glidelive_pocket SHARED pocket_tts.cpp pocket_jni.cpp)
target_compile_features(glidelive_pocket PRIVATE cxx_std_17)
target_compile_options(glidelive_pocket PRIVATE -O2 -fvisibility=hidden)
target_compile_definitions(glidelive_pocket PRIVATE PTT_SHARED_LIB)
target_include_directories(glidelive_pocket PRIVATE ort dr_libs sentencepiece/src)
target_link_options(glidelive_pocket PRIVATE "-Wl,--exclude-libs,ALL")
target_link_libraries(glidelive_pocket PRIVATE "${ORT_LIBRARY}" sentencepiece-static log android)
